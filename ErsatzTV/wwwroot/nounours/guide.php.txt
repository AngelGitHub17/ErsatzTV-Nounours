<?php
/*
 * ErsatzTV – édition Nounours : page « programme TV » simple, à partir de guide.json
 * (envoyé par recevoir-guide.php). À adapter à la charte de votre site.
 */
$guide = json_decode(@file_get_contents(__DIR__ . '/guide.json'), true) ?: ['chaines' => []];
function h($s) { return htmlspecialchars((string)$s, ENT_QUOTES, 'UTF-8'); }
function heure($iso) { return (new DateTime($iso))->format('H\hi'); }
?>
<!doctype html>
<html lang="fr">
<head>
    <meta charset="utf-8">
    <meta name="viewport" content="width=device-width, initial-scale=1">
    <title>Programme TV</title>
    <style>
        body { background: #111; color: #eee; font-family: system-ui, sans-serif; margin: 0; padding: 24px; }
        h1 { font-style: italic; text-transform: uppercase; }
        .chaines { display: grid; gap: 18px; grid-template-columns: repeat(auto-fill, minmax(300px, 1fr)); }
        .chaine { background: #1c1f27; border-radius: 12px; padding: 16px; }
        .chaine h2 { align-items: center; display: flex; font-size: 1.1rem; gap: 10px; margin: 0 0 12px; }
        .chaine h2 img { height: 32px; }
        .encours { background: #ff4000; border-radius: 8px; color: #fff; margin-bottom: 10px; padding: 8px 10px; }
        .ligne { border-top: 1px solid #333; display: flex; gap: 10px; padding: 6px 0; }
        .ligne b { flex: none; width: 52px; }
        a.regarder { color: #4f8cff; font-size: .9rem; }
    </style>
</head>
<body>
<h1>Programme TV</h1>
<div class="chaines">
    <?php foreach ($guide['chaines'] as $c): ?>
        <div class="chaine">
            <h2><?php if (!empty($c['logo'])): ?><img src="<?= h($c['logo']) ?>" alt=""><?php endif ?><?= h($c['numero']) ?> · <?= h($c['nom']) ?></h2>
            <?php if (!empty($c['enCours'])): ?>
                <div class="encours">En ce moment : <b><?= h($c['enCours']['titre']) ?></b> (jusqu'à <?= heure($c['enCours']['fin']) ?>)</div>
            <?php endif ?>
            <?php foreach (array_slice($c['programmes'] ?? [], 0, 12) as $p): ?>
                <div class="ligne"><b><?= heure($p['debut']) ?></b><span><?= h($p['titre']) ?><?= !empty($p['sousTitre']) ? ' – ' . h($p['sousTitre']) : '' ?></span></div>
            <?php endforeach ?>
            <?php if (!empty($c['flux']['hls'])): ?><a class="regarder" href="<?= h($c['flux']['hls']) ?>">Regarder (lien du flux)</a><?php endif ?>
        </div>
    <?php endforeach ?>
</div>
<p style="opacity:.6">Mis à jour le <?= !empty($guide['genereLe']) ? (new DateTime($guide['genereLe']))->format('d/m/Y à H\hi') : '—' ?></p>
</body>
</html>
